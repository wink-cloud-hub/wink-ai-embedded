# devkitc_smoke: ESP32 bare-metal S1-S8 full-chain smoke (host e2e verifies
# code structure; real signal is on hardware). Shared boilerplate in
# ../sample_common.cmake.
include(${CMAKE_CURRENT_LIST_DIR}/../sample_common.cmake)

# ── wasm target hard reject (2026-07-01 wasm simulator repair plan Task 1.3) ──
# test_devkitc_smoke_e2e.c pulls host_test_ctrl.h (host-only) and the business
# logic depends on ESP32 GPIO interrupt behavior. FATAL_ERROR beats silent
# return() — the caller learns they picked the wrong App.
if(EMSCRIPTEN)
    message(FATAL_ERROR
        "Sample 'devkitc_smoke' is a physical hardware smoke test and is not "
        "supported on the WebAssembly simulator target. Use "
        "-DWINK_APP_DIR=<...>/samples/avoidance_car (or oled_dashboard) for wasm.")
endif()

# ── Codegen: wink-app.json → device_tree.[hc] + app_options.cmake ──
# P1-4 (2026-07-06): devkitc_smoke migrates to declarative device tree. The
# hand-written device_tree.c/h were deleted; the three artefacts below are
# regenerated by tools/codegen/app_codegen.py every time wink-app.json changes.
# (app_support.c was retired in the BAL refactor; callbacks now live in
# app_callbacks.c directly.)
# Python3 is already found at the top-level CMakeLists.txt (line 20), so we
# just reuse ${Python3_EXECUTABLE} here.
set(WINK_APP_JSON     ${CMAKE_CURRENT_SOURCE_DIR}/wink-app.json)
if(NOT DEFINED WINK_CODEGEN_ROOT)
    set(WINK_CODEGEN_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/tools/codegen)
endif()
set(WINK_CODEGEN_TOOL ${WINK_CODEGEN_ROOT}/app_codegen.py)
set(GEN_DIR           ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})

set(GEN_DEVICE_TREE_H ${GEN_DIR}/device_tree.h)
set(GEN_DEVICE_TREE_C ${GEN_DIR}/device_tree.c)
set(GEN_APP_OPTIONS   ${GEN_DIR}/app_options.cmake)

# Re-run CMake configure when wink-app.json changes so downstream conditional
# logic (P2: app_options.cmake include) picks up new options.
set_property(DIRECTORY ${CMAKE_SOURCE_DIR} APPEND
     PROPERTY CMAKE_CONFIGURE_DEPENDS ${WINK_APP_JSON})

# Any change to a codegen template or driver plugin must retrigger generation.
file(GLOB WINK_CODEGEN_TEMPLATES ${WINK_CODEGEN_ROOT}/templates/*.j2)
file(GLOB WINK_CODEGEN_DRIVERS   ${WINK_CODEGEN_ROOT}/drivers/*.py)

add_custom_command(
    OUTPUT ${GEN_DEVICE_TREE_H} ${GEN_DEVICE_TREE_C}
           ${GEN_APP_OPTIONS}
    COMMAND ${Python3_EXECUTABLE} ${WINK_CODEGEN_TOOL}
            --config ${WINK_APP_JSON}
            --out-dir ${GEN_DIR}
    DEPENDS ${WINK_APP_JSON} ${WINK_CODEGEN_TOOL}
            ${WINK_CODEGEN_TEMPLATES} ${WINK_CODEGEN_DRIVERS}
    COMMENT "Generating devkitc_smoke device tree from wink-app.json"
    VERBATIM
)

set(APP_SOURCES
    ${GEN_DEVICE_TREE_C}
    ${CMAKE_CURRENT_SOURCE_DIR}/app_callbacks.c)

# host e2e binary: runtime + trace + dal_led + dal_button + dal_ultrasonic + pal_host + app + selftest + common helpers
set(WINK_SELFTEST_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_core.c
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_pwm_router.c
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_i2c_scan.c
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_smp_stress.c
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_gpio_isr.c
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/selftest_rmt_loopback.c
    # ADR-0023 Stage 2.4: S10 ultrasonic echo sim bringup helper (migrated
    # from samples/common/). Strict-nonblocking gate lives in the TU itself.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src/wink_sim_ultrasonic_echo.c
)

add_executable(app_devkitc_smoke_e2e
    ${APP_SOURCES}
    ${WINK_SAMPLE_RUNTIME_SOURCES}
    ${WINK_SELFTEST_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/test_devkitc_smoke_e2e.c
    $<TARGET_OBJECTS:pal_host>)
wink_sample_apply_common(app_devkitc_smoke_e2e)
wink_sample_link_common(app_devkitc_smoke_e2e)
# P2 (2026-07-06): link the dal static library instead of compiling
# dal/src/**/*.c directly. This makes WINK_USE_XXX pruning options
# (optionally driven by the generated app_options.cmake) take effect.
target_link_libraries(app_devkitc_smoke_e2e PRIVATE dal)
target_compile_definitions(app_devkitc_smoke_e2e PRIVATE WINK_CFG_SIM_ECHO=1)
# Selftest internal headers (registry.def / wink_selftest_internal.h) live
# alongside the sources — app doesn't include them but the selftest .c files do.
# GEN_DIR: generated device_tree.h (codegen output) lives in the build tree
# (app_support.c was retired in the 2026-07-06 BAL refactor; callbacks now
# live directly in app_callbacks.c).
target_include_directories(app_devkitc_smoke_e2e PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../wink-micro-os/runtime/selftest/src
    ${GEN_DIR})
add_test(NAME app_devkitc_smoke_e2e COMMAND app_devkitc_smoke_e2e)
